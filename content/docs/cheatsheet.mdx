Type-safe HTTP APIs, for developers and agents. Write a route once with the
route builder, and the same route is a validated HTTP endpoint, a typed fetch
client, TanStack Query options, an agent tool, and an in-process call for
tests. Below are the common patterns, each a complete example. See the
[documentation](/docs/quickstart) for the rest.
For agents, [llms.txt](https://rest-rpc.vercel.app/llms.txt) lists every
docs page and the ways to fetch them.

| Package                                                                         | Contains                                                                        |
| ------------------------------------------------------------------------------- | ------------------------------------------------------------------------------- |
| [`@rest-rpc/core`](https://npmx.dev/package/@rest-rpc/core)                     | Contracts, Fetch client, contract extraction, OpenAPI                           |
| [`@rest-rpc/server`](https://npmx.dev/package/@rest-rpc/server)                 | Route builder and handlers for Fetch runtimes, `@rest-rpc/server/node` for Node |
| [`@rest-rpc/tanstack-query`](https://npmx.dev/package/@rest-rpc/tanstack-query) | TanStack Query options and keys                                                 |

Server code, its tests included, imports from `@rest-rpc/server`, which
re-exports what it needs from `@rest-rpc/core`. Client code imports
`@rest-rpc/core`.

## Hello World

A route, an app and a typed client. The client here calls the app in the same
process, which is also how to test it.

See [Quickstart](/docs/quickstart)

```ts
import { createApp, createClient, route } from "@rest-rpc/server";

const router = {
	hello: route.get("/hello").handler(() => ({ message: "Hello" })),
};

const app = createApp(router); // `export default app` to serve it on Bun, Deno or Workers

const client = createClient(router, {
	origin: "http://localhost",
	fetch: app.fetch,
});
const { status, body } = await client.hello();
console.log(status, body.message); // 200 "Hello", typed from the handler
```

## Methods and Paths

A route without a method is a `POST` at its keys in the route tree. `:name`
segments are string params.

See [Route Builder](/docs/route-builder#choose-a-method-and-path)

```ts
import { route } from "@rest-rpc/server";
import * as z from "zod";

const base = route.prefix("/api");

export const router = {
	posts: {
		list: base
			.get("/posts")
			.handler(() => [{ id: "1", title: "Hello", body: "First post" }]), // GET /api/posts
		get: base.get("/posts/:id").handler(({ params }) => ({ id: params.id })), // params.id: string
		create: base
			.body(z.object({ title: z.string(), body: z.string() }))
			.handler(({ body }) => ({ id: "2", title: body.title, body: body.body })), // POST /api/posts/create
	},
};
```

## Request Validation

Params, query, headers and body are validated before the handler runs, with
any [Standard Schema](https://standardschema.dev) such as Zod, Valibot or
ArkType, and an invalid request gets a `400`. Params, query and headers arrive
as strings, so their schemas coerce.

See [Validation](/docs/http-behavior/validation)

```ts
import { route } from "@rest-rpc/server";
import * as z from "zod";

const search = route
	.get("/users/:id/posts")
	.params(z.object({ id: z.coerce.number() }))
	.query(
		z.object({
			sort: z.enum(["new", "top"]).optional(),
			tags: z.array(z.string()).optional(),
		}),
	)
	.headers(z.object({ "x-request-id": z.string() }))
	.handler(({ params, query, headers }) => ({
		userId: params.id, // number
		sort: query.sort ?? "new",
		tags: query.tags ?? [], // sent as tags[]=a&tags[]=b
		requestId: headers["x-request-id"],
	}));
```

## Status, Headers and Cookies

Return the body, or `respond()` to set the rest of the response. The client
receives them with lowercase header names.

See [Return Responses](/docs/route-builder#return-responses)

```ts
import { createApp, createClient, route } from "@rest-rpc/server";
import * as z from "zod";

const create = route
	.post("/posts")
	.body(z.object({ title: z.string(), body: z.string() }))
	.handler(({ body, respond }) => {
		const post = {
			id: crypto.randomUUID(),
			title: body.title,
			body: body.body,
		};
		return respond(post, {
			status: 201,
			headers: { location: `/posts/${post.id}` },
			cookies: ["draft=; Max-Age=0"],
		});
	});

const router = { create };

const app = createApp(router);
const client = createClient(router, {
	origin: "http://localhost",
	fetch: app.fetch,
});

const { status, headers } = await client.create({
	body: { title: "Hello", body: "First post" },
});
console.log(status, headers.location); // 201 "/posts/..."
```

## Defined Responses

Responses are inferred from the handler. Define them to validate them at
runtime and document them in OpenAPI.

See [Define Responses](/docs/route-builder#define-responses)

```ts
import { route } from "@rest-rpc/server";
import * as z from "zod";

const post = z.object({ id: z.string(), title: z.string(), body: z.string() });

const create = route
	.post("/posts")
	.body(z.object({ title: z.string(), body: z.string() }))
	.response(post, { status: 201, headers: z.object({ location: z.string() }) })
	.handler(({ body, respond }) => {
		const created = {
			id: crypto.randomUUID(),
			title: body.title,
			body: body.body,
		};
		return respond(created, {
			status: 201,
			headers: { location: `/posts/${created.id}` },
		});
	});
```

## Errors

Errors are thrown. A defined error is typed on the client, and any other
`RouteError` is still sent with its status.

See [Errors](/docs/http-behavior/errors)

```ts
import {
	createApp,
	createClient,
	isDefinedError,
	route,
	safe,
} from "@rest-rpc/server";
import * as z from "zod";

const posts = new Map([["1", { id: "1", title: "Hello", body: "First post" }]]);

const router = {
	get: route
		.get("/posts/:id")
		.error("POST_NOT_FOUND", {
			status: 404,
			data: z.object({ id: z.string() }),
		})
		.handler(({ params, errors }) => {
			const post = posts.get(params.id);
			if (!post) throw errors.POST_NOT_FOUND({ data: { id: params.id } });
			return post;
		}),
};

const app = createApp(router);
const client = createClient(router, {
	origin: "http://localhost",
	fetch: app.fetch,
});

const { error, data } = await safe(client.get({ params: { id: "2" } }));
if (isDefinedError(error)) console.log(error.code, error.data.id); // "POST_NOT_FOUND" "2"
else if (error) throw error;
else console.log(data.body.title);
```

## Shared Errors

`routeError()` defines an error once, to throw it from any code and add it to
routes.

See [Define Errors](/docs/route-builder#define-errors)

```ts
import { route, routeError } from "@rest-rpc/server";
import * as z from "zod";

const PostNotFound = routeError("POST_NOT_FOUND", {
	status: 404,
	data: z.object({ id: z.string() }),
});

const posts = new Map([["1", { id: "1", title: "Hello", body: "First post" }]]);

const findPost = (id: string) => {
	const post = posts.get(id);
	if (!post) throw new PostNotFound({ data: { id } });
	return post;
};

const get = route
	.get("/posts/:id")
	.error(PostNotFound)
	.handler(({ params }) => findPost(params.id));
```

## Middleware, Context and Metadata

Middleware runs after validation, around the handler, and passes typed values
to it through context. Initial context comes from the host, and routes describe
themselves to shared middleware with metadata of a declared shape.

See [Middleware](/docs/middleware)

```ts
import { createApp, route } from "@rest-rpc/server";

const base = route
	.$meta<{ access: "public" | "signed-in" }>()
	.$context<{ headers: Headers }>() // provided by the app below
	.error("UNAUTHORIZED", { status: 401 })
	.use(({ route, context, errors, next }) => {
		const user = context.headers.get("authorization");
		if (!user && route.meta?.access !== "public") throw errors.UNAUTHORIZED();
		return next({ context: { user } });
	});

const router = {
	health: base
		.get("/health")
		.meta({ access: "public" })
		.handler(() => "ok"),
	me: base
		.get("/me")
		.meta({ access: "signed-in" })
		.handler(({ context }) => context.user), // string | null
};

export default createApp(router, {
	context: (request) => ({ headers: request.headers }), // or an object, such as { db }
});
```

## Server-Sent Events

Return an async iterable to stream events. The client iterates them.

See [Server-Sent Events](/docs/server-sent-events)

```ts
import {
	createApp,
	createClient,
	route,
	sse,
	sseEvent,
} from "@rest-rpc/server";
import * as z from "zod";

const messages = ["Ada liked your post", "Grace commented on your post"];

const router = {
	notifications: route
		.get("/notifications")
		.response(sse(z.object({ message: z.string() }))) // optional, validates each event
		.handler(async function* ({ signal }) {
			for (const [i, message] of messages.entries()) {
				if (signal.aborted) break;
				yield sseEvent({ message }, { id: String(i) });
			}
		}),
};

const app = createApp(router);
const client = createClient(router, {
	origin: "http://localhost",
	fetch: app.fetch,
});

const { body } = await client.notifications();
for await (const event of body) console.log(event.id, event.data.message);
```

## Content Types

A body is JSON unless a `content-type` header says otherwise, and `FormData`
is always `multipart/form-data`. The header's media type selects one of these
built-in codecs, each exported the same way from `@rest-rpc/server` and
`@rest-rpc/core`.

See [Encoding and Content Types](/docs/http-behavior/encoding#built-in-codecs)

```ts
import {
	jsonCodec, // application/json and +json: any value JSON.stringify encodes
	urlEncodedCodec, // application/x-www-form-urlencoded: URLSearchParams
	formDataCodec, // multipart/form-data: FormData
	textCodec, // text/*: strings, decoded as UTF-8
	binaryCodec, // any other media type: Blob, Uint8Array, ReadableStream or async iterable, received as a Blob
} from "@rest-rpc/server";
```

## Codecs

A codec encodes another format. Pass the same codec in `bodyCodecs` to the
client and the server. Custom codecs are selected before the built-in ones.

See [Custom Codecs](/docs/http-behavior/encoding#custom-codecs)

```ts
import {
	binaryCodec,
	bodyCodec,
	createApp,
	createClient,
	route,
	serializedBody,
} from "@rest-rpc/server";
import superjson from "superjson";

const superjsonCodec = bodyCodec("application/vnd.superjson+json", {
	serialize: (value) => superjson.stringify(value),
	deserialize: async (body) => superjson.parse(await body.text()),
});

const bytes = bodyCodec("application/octet-stream", {
	serialize: (value) =>
		value instanceof Uint8Array
			? serializedBody(value, { headers: { "x-body": "uint8array" } })
			: binaryCodec.serialize(value),
	deserialize: async (body, headers) =>
		headers["x-body"] === "uint8array"
			? new Uint8Array(await body.arrayBuffer())
			: binaryCodec.deserialize(body),
});

const router = {
	posts: {
		get: route
			.get("/posts/:id")
			.handler(({ params, respond }) =>
				respond(
					{ id: params.id, publishedAt: new Date() },
					{ headers: { "content-type": "application/vnd.superjson+json" } },
				),
			),
		export: route.get("/posts/:id/export").handler(({ respond }) =>
			respond(new TextEncoder().encode("Hello"), {
				headers: { "content-type": "application/octet-stream" },
			}),
		),
	},
};

const bodyCodecs = [superjsonCodec, bytes];
const app = createApp(router, { bodyCodecs });
const client = createClient(router, {
	origin: "http://localhost",
	fetch: app.fetch,
	bodyCodecs,
});

const { body: post } = await client.posts.get({ params: { id: "post_1" } });
post.publishedAt; // a Date
const { body: exported } = await client.posts.export({
	params: { id: "post_1" },
});
exported; // a Uint8Array
```

## Raw and Static Responses

Return a Fetch `Response` for anything else, such as a redirect. A route that
always sends the same response is `.static()`, built once.

See [Raw Responses](/docs/route-builder#raw-responses)

```ts
import { route } from "@rest-rpc/server";

const download = route
	.get("/images/:id")
	.handler(({ params }) =>
		Response.redirect(`https://cdn.example.com/${params.id}`, 302),
	);

const health = route.get("/health").static({ ok: true });
```

## Interceptors

Interceptors run around every request the app serves, for what rest-rpc
leaves to the app, such as API keys, CORS and error reporting.

See [Interceptors](/docs/server/interceptors)

```ts
import {
	createApp,
	errorResponse,
	interceptor,
	onError,
	route,
} from "@rest-rpc/server";

const requestId = interceptor(async ({ next }) => {
	const response = await next();
	const copy = new Response(response.body, response); // its headers may be immutable
	copy.headers.set("x-request-id", crypto.randomUUID());
	return copy;
});

const router = { hello: route.get("/hello").handler(() => "Hello") };

export default createApp(router, {
	interceptors: [
		requestId,
		interceptor(({ next }) => next().catch(errorResponse)), // errors reach requestId as responses
		onError((error) => console.error(error)), // errors before the response and in event streams
	],
});
```

## Frameworks

Mount the routes in a framework under one route, or register each on the
framework's router.

See [Framework Integration](/docs/server/framework-integration)

```ts
import { Hono } from "hono";
import { createCatchAllHandler, route } from "@rest-rpc/server";

const router = { hello: route.get("/hello").handler(() => "Hello") };
const handler = createCatchAllHandler(router);

const app = new Hono();
app.all("/api/*", async (c) => {
	const result = await handler(c.req.raw, { mountPath: "/api" });
	return result.matched ? result.response : c.notFound();
});

export default app;
```

## Browser Clients

A bundler macro runs `extractContract()` at build time and inlines the routes'
methods, paths, statuses and error codes, so no server code ships. Bun runs
macros itself, and Vite and other bundlers with `unplugin-macros`. The client
sections below use these files.

See [Contract Extraction](/docs/client/contract-extraction)

```ts router.ts
import { route } from "@rest-rpc/server";
import * as z from "zod";

const posts = new Map([["1", { id: "1", title: "Hello", body: "First post" }]]);

export const router = {
	posts: {
		list: route.get("/posts").handler(() => [...posts.values()]),
		get: route
			.get("/posts/:id")
			.error("POST_NOT_FOUND", { status: 404 })
			.handler(({ params, errors }) => {
				const post = posts.get(params.id);
				if (!post) throw errors.POST_NOT_FOUND();
				return post;
			}),
		create: route
			.body(z.object({ title: z.string(), body: z.string() }))
			.handler(({ body }) => {
				const post = { id: crypto.randomUUID(), ...body };
				posts.set(post.id, post);
				return post;
			}),
	},
};
```

```ts contract.ts
import { extractContract } from "@rest-rpc/core";
import { router } from "./router.ts";

export function getContract() {
	return extractContract(router);
}
```

```ts client.ts
import { createClient } from "@rest-rpc/core";
import { getContract } from "./contract.ts" with { type: "macro" };

export const client = createClient(getContract()); // the page's origin, or pass { origin }
```

```ts vite.config.ts
import Macros from "unplugin-macros/vite";
import { defineConfig } from "vite";

export default defineConfig({ plugins: [Macros()] });
```

## Client Options

Headers for every request, the server's mount path, and interceptors around
each request.

See [Fetch Client](/docs/client/fetch-client#client-options)

```ts client.ts
import { clientInterceptor, createClient } from "@rest-rpc/core";
import { getContract } from "./contract.ts" with { type: "macro" };

const logging = clientInterceptor(async ({ request, next }) => {
	const response = await next();
	console.log(request.method, request.url, response.status);
	return response;
});

export const client = createClient(getContract(), {
	origin: "https://api.example.com",
	mountPath: "/api",
	globalHeaders: {
		authorization: () => `Bearer ${localStorage.getItem("token")}`,
	},
	interceptors: [logging],
});
```

## TanStack Query

Query and mutation options for each route, with typed keys.

See [TanStack Query](/docs/client/tanstack-query)

```ts client.ts
import { createClient } from "@rest-rpc/core";
import { createTanstackQueryUtils } from "@rest-rpc/tanstack-query";
import { getContract } from "./contract.ts" with { type: "macro" };

export const client = createClient(getContract());
export const queryUtils = createTanstackQueryUtils(client);
```

```tsx posts.tsx
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { queryUtils } from "./client";

export function Posts() {
	const queryClient = useQueryClient();
	const posts = useQuery(
		queryUtils.posts.list.queryOptions({ select: (response) => response.body }),
	);
	const create = useMutation(
		queryUtils.posts.create.mutationOptions({
			onSuccess: () =>
				queryClient.invalidateQueries({
					queryKey: queryUtils.posts.list.key(),
				}),
		}),
	);
	return (
		<button
			onClick={() =>
				create.mutate({ body: { title: "Hello", body: "First post" } })
			}
		>
			{posts.data?.length ?? 0} posts
		</button>
	);
}
```

## Contract-first

Define the routes in a shared contract with `@rest-rpc/core`, and implement
them on the server. The client uses the contract directly, schemas included.

See [Contract-first](/docs/contract-first)

```ts contract.ts
import { route } from "@rest-rpc/core";
import * as z from "zod";

export const contract = {
	posts: {
		get: route
			.get("/posts/:id")
			.response(
				z.object({ id: z.string(), title: z.string(), body: z.string() }),
			)
			.error("POST_NOT_FOUND", { status: 404 }),
	},
};
```

```ts server.ts
import { createApp, implement } from "@rest-rpc/server";
import { contract } from "./contract.ts";

const router = {
	posts: {
		get: implement(contract.posts.get).handler(({ params }) => ({
			id: params.id,
			title: "Hello",
			body: "First post",
		})),
	},
};

export default createApp(router);
```

```ts client.ts
import { createClient } from "@rest-rpc/core";
import { contract } from "./contract";

export const client = createClient(contract, { validateResponses: true });
```

## Local Calls

`call()` runs one route in process, through validation and middleware, without
HTTP. To call the app in process with HTTP behavior, pass `app.fetch` to
`createClient` as in [Hello World](#hello-world).

See [Local Calls](/docs/client/local-calls)

```ts
import { call, route } from "@rest-rpc/server";
import * as z from "zod";

const double = route
	.get("/double/:n")
	.params(z.object({ n: z.coerce.number() }))
	.handler(({ params }) => params.n * 2);

const { body } = await call(double, { params: { n: 21 } });
console.log(body); // 42
```

## OpenAPI

An OpenAPI document from the same routes, with a converter for your schema
library.

See [OpenAPI](/docs/openapi)

```ts
import { createOpenApiDocument, route } from "@rest-rpc/server";
import * as z from "zod";

const router = {
	posts: {
		list: route
			.get("/posts")
			.response(
				z.array(
					z.object({ id: z.string(), title: z.string(), body: z.string() }),
				),
			)
			.openAPI({ summary: "List posts", tags: ["Posts"] })
			.handler(() => [{ id: "1", title: "Hello", body: "First post" }]),
	},
};

const document = createOpenApiDocument(router, {
	info: { title: "Blog API", version: "1.0.0" },
	schemaConverter: (schema, io) => z.toJSONSchema(schema as z.ZodType, { io }),
});
```

## MCP Tools

`createToolsFromRoutes()` turns routes into tools, to register on an MCP server
or in an agent framework. `mergeTools()` merges them into two tools for an API
with many routes.

See [MCP Tools](/docs/mcp-tools)

```ts
import {
	createMcpHandler,
	fromJsonSchema,
	McpServer,
} from "@modelcontextprotocol/server";
import { createApp, createToolsFromRoutes, route } from "@rest-rpc/server";
import * as z from "zod";

const post = z.object({ id: z.string(), title: z.string(), body: z.string() });
const posts = [{ id: "1", title: "Hello", body: "First post" }];

const router = {
	posts: {
		search: route
			.get("/posts/search")
			.query(z.object({ text: z.string().describe("Words to find in posts") }))
			.response(z.object({ posts: z.array(post) }))
			.openAPI({ summary: "Search posts by their title and body" })
			.handler(({ query }) => ({
				posts: posts.filter(({ title, body }) =>
					`${title} ${body}`.includes(query.text),
				),
			})),
	},
};

const tools = createToolsFromRoutes(router, {
	schemaConverter: (schema, io) => z.toJSONSchema(schema as z.ZodType, { io }),
});

const mcp = createMcpHandler(() => {
	const server = new McpServer({ name: "blog", version: "1.0.0" });
	for (const { definition, call } of tools) {
		const { name, inputSchema, outputSchema, ...config } = definition;
		server.registerTool(
			name,
			{
				...config,
				inputSchema: fromJsonSchema(inputSchema),
				...(outputSchema && { outputSchema: fromJsonSchema(outputSchema) }),
			},
			(args) => call(args),
		);
	}
	return server;
});

const app = createApp(router);

export default {
	fetch: (request: Request) =>
		new URL(request.url).pathname === "/mcp"
			? mcp.fetch(request)
			: app.fetch(request),
};
```

## Type Helpers

Reuse a route's types in code outside it, such as a service written apart
from its route.

See [Type Helpers](/docs/route-builder#type-helpers)

```ts
import type { InferClientResponse } from "@rest-rpc/core";
import { route, type InferServerRequest } from "@rest-rpc/server";
import * as z from "zod";

const updateDefinition = route
	.patch("/posts/:id")
	.body(z.object({ title: z.string(), body: z.string() }));

const updatePost = ({
	params,
	body,
}: InferServerRequest<typeof updateDefinition>) => ({
	id: params.id,
	title: body.title,
	body: body.body,
});

const router = { posts: { update: updateDefinition.handler(updatePost) } };

type UpdateResponse = InferClientResponse<typeof router.posts.update>; // { status: 200; body: { id; title; body }; ... }
```
