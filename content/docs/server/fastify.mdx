## Install

```sh
pnpm add @rest-rpc/fastify
```

## Usage

```ts
import { implement, registerRoutes } from "@rest-rpc/fastify";
import Fastify from "fastify";
import { api } from "./contract";

const app = Fastify();

const implementor = implement(api);

const routes = {
	todos: {
		list: implementor.todos.list.handler(() => ({
			status: 200,
			body: listTodos(),
		})),
		get: implementor.todos.get.handler(({ params: { id }, req }) => {
			const authorization = req.headers.authorization;
			return { status: 200, body: getTodo(id, { authorization }) };
		}),
		create: implementor.todos.create.handler(({ body: { title } }) => {
			return { status: 201, body: createTodo({ title }) };
		}),
	},
};

registerRoutes(app, routes);

await app.listen({ port: 3000 });
```

## Framework Context

Handlers receive the following adapter fields directly:

```ts
type RouteHandlerFields = {
	req: FastifyRequest;
	reply: FastifyReply;
	signal: AbortSignal;
};
```

## Options

```ts
type RegisterRoutesOptions = {
	requestValidationErrorHandler?: RequestValidationErrorHandler;
	responseValidationErrorHandler?: ResponseValidationErrorHandler;
	preHandler?: Array<
		(
			req: FastifyRequest,
			reply: FastifyReply,
			route: RouteDeclaration,
		) => unknown
	>;
};
```

### Error handling

Request validation errors default to a 400 response. Response contract
validation errors default to a generic 500 response. Customize them with
handlers that receive native Fastify arguments. Other errors are handled by
Fastify's error handler.

```ts
registerRoutes(app, routes, {
	requestValidationErrorHandler: (error, request, reply) => {
		return reply.status(422).send({
			code: "VALIDATION_ERROR",
			issues: error.issues,
		});
	},
	responseValidationErrorHandler: (_error, request, reply) => {
		return reply.status(500).send({ code: "INVALID_RESPONSE" });
	},
});

app.setErrorHandler((error, request, reply) => {
	return reply.status(500).send({ code: "INTERNAL_SERVER_ERROR" });
});
```

### preHandler

```ts
registerRoutes(app, routes, {
	preHandler: [
		async (req, reply, route) => {
			const authorization = req.headers.authorization;
			if (!authorization) {
				reply.status(401).send({ code: "UNAUTHORIZED" });
			}
		},
	],
});
```
